cmake_minimum_required(VERSION 3.12)
project(bspai_cpu_debug LANGUAGES CXX)
include(CTest)

add_library(bspai_cpu src/cpu.cpp src/spai.cpp src/gmres.cpp src/ir.cpp)
target_include_directories(bspai_cpu PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(bspai_cpu PUBLIC cxx_std_11)
if(MSVC)
  target_compile_options(bspai_cpu PRIVATE /W4)
  target_compile_options(bspai_cpu PUBLIC /fp:strict)
else()
  target_compile_options(bspai_cpu PUBLIC -fno-fast-math -ffp-contract=off)
  target_compile_options(bspai_cpu PRIVATE -Wall -Wextra -Wpedantic)
  if(MINGW AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_SIZEOF_VOID_P EQUAL 4)
    # Enforce binary64 intermediates instead of x87 excess precision.
    target_compile_options(bspai_cpu PUBLIC -msse2 -mfpmath=sse)
  endif()
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_spai.cpp")
  add_executable(spai_tests tests/test_spai.cpp)
  target_link_libraries(spai_tests PRIVATE bspai_cpu)
  add_test(NAME spai_tests COMMAND spai_tests)
endif()

add_executable(cpu_matrix_probe src/main.cpp)
target_link_libraries(cpu_matrix_probe PRIVATE bspai_cpu)

if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cpu.cpp")
  add_executable(cpu_core_tests tests/test_cpu.cpp)
  target_link_libraries(cpu_core_tests PRIVATE bspai_cpu)
  add_test(NAME cpu_core_tests COMMAND cpu_core_tests)
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_gmres.cpp")
  add_executable(gmres_tests tests/test_gmres.cpp)
  target_link_libraries(gmres_tests PRIVATE bspai_cpu)
  add_test(NAME gmres_tests COMMAND gmres_tests)
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ir.cpp")
  add_executable(ir_tests tests/test_ir.cpp)
  target_link_libraries(ir_tests PRIVATE bspai_cpu)
  add_test(NAME ir_tests COMMAND ir_tests)
endif()
