cmake_minimum_required(VERSION 3.18)
project(spai_gmres_ir LANGUAGES CXX)
include(CTest)
option(BSPAI_ENABLE_CUDA "Build CUDA FP32/FP64 solvers" OFF)
option(BSPAI_BUILD_CPU_TESTS "Build CPU regression tests" ON)
set(BSPAI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")

# One arithmetic policy for the libraries and their callers.
add_library(bspai_options INTERFACE)
target_compile_features(bspai_options INTERFACE cxx_std_11)
target_include_directories(bspai_options INTERFACE "${BSPAI_ROOT}/include")
if(MSVC)
  target_compile_options(bspai_options INTERFACE
    $<$<COMPILE_LANGUAGE:CXX>:/W4;/fp:strict>
    $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/W4,/fp:strict>)
else()
  target_compile_options(bspai_options INTERFACE
    $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-fno-fast-math;-ffp-contract=off>)
endif()
target_compile_options(bspai_options INTERFACE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)

function(bspai_library target)
  add_library(${target} ${ARGN})
  target_link_libraries(${target} PUBLIC bspai_options)
endfunction()
function(bspai_executable target source library)
  add_executable(${target} "${BSPAI_ROOT}/${source}")
  target_link_libraries(${target} PRIVATE ${library})
endfunction()
function(bspai_test target source library)
  bspai_executable(${target} "${source}" ${library})
  add_test(NAME ${target} COMMAND ${target})
endfunction()

bspai_library(bspai_cpu src/cpu.cpp src/spai.cpp src/gmres.cpp src/ir.cpp)
if(MINGW AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_SIZEOF_VOID_P EQUAL 4)
  # Retain the original FP64 x87-excess-precision guard on its original target.
  target_compile_options(bspai_cpu PUBLIC $<$<COMPILE_LANGUAGE:CXX>:-msse2;-mfpmath=sse>)
endif()
bspai_library(bspai_fp32_cpu src/fp32_cpu.cpp)
bspai_executable(cpu_matrix_probe src/main.cpp bspai_cpu)
bspai_executable(cpu_fp64_ir src/solver_main.cpp bspai_cpu)
bspai_executable(cpu_fp32_ir src/solver_main.cpp bspai_fp32_cpu)
target_compile_definitions(cpu_fp32_ir PRIVATE BSPAI_SOLVER_FP32=1)
# The FP32 CLI shares the input parser; the solver itself stays binary32.
target_link_libraries(cpu_fp32_ir PRIVATE bspai_cpu)
if(BUILD_TESTING)
  bspai_executable(cpu_release_snapshot tests/release_cpu_snapshot.cpp bspai_cpu)
  if(BSPAI_BUILD_CPU_TESTS)
    bspai_test(cpu_core_tests tests/test_cpu.cpp bspai_cpu)
    bspai_test(spai_tests tests/test_spai.cpp bspai_cpu)
    bspai_test(gmres_tests tests/test_gmres.cpp bspai_cpu)
    bspai_test(ir_tests tests/test_ir.cpp bspai_cpu)
    bspai_test(fp32_cpu_tests tests/test_fp32.cpp bspai_fp32_cpu)
  endif()
endif()
if(BSPAI_ENABLE_CUDA)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 11)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  find_package(CUDAToolkit REQUIRED)
  include(cmake/CudaTargets.cmake)
endif()
