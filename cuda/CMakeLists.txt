cmake_minimum_required(VERSION 3.18)
project(bspai_cuda_spmv LANGUAGES CXX CUDA)
include(CTest)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "CUDA milestone requires a separate 64-bit host build")
endif()
find_package(CUDAToolkit REQUIRED)
add_library(bspai_cuda_spmv ../src/cpu.cpp ../src/cuda_spmv.cu)
target_include_directories(bspai_cuda_spmv PUBLIC ../include)
target_compile_features(bspai_cuda_spmv PUBLIC cxx_std_11)
set_target_properties(bspai_cuda_spmv PROPERTIES CUDA_STANDARD 11 CUDA_STANDARD_REQUIRED ON)
target_link_libraries(bspai_cuda_spmv PUBLIC CUDA::cudart CUDA::cusparse)
add_library(bspai_cuda_gmres ../src/gmres.cpp ../src/spai.cpp ../src/cuda_gmres.cu)
target_link_libraries(bspai_cuda_gmres PUBLIC bspai_cuda_spmv CUDA::cublas)
target_compile_features(bspai_cuda_gmres PUBLIC cxx_std_11)
set_target_properties(bspai_cuda_gmres PROPERTIES CUDA_STANDARD 11 CUDA_STANDARD_REQUIRED ON)
if(MSVC)
  target_compile_options(bspai_cuda_gmres PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/fp:strict> $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/W4,/fp:strict>)
endif()
target_compile_options(bspai_cuda_gmres PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
add_library(bspai_cuda_spai ../src/cuda_spai.cu)
target_link_libraries(bspai_cuda_spai PUBLIC bspai_cuda_gmres CUDA::cusolver)
set_target_properties(bspai_cuda_spai PROPERTIES CUDA_STANDARD 11 CUDA_STANDARD_REQUIRED ON)
target_compile_options(bspai_cuda_spai PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
if(MSVC)
  target_compile_options(bspai_cuda_spai PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/W4,/fp:strict>)
endif()
add_library(bspai_cuda_ir ../src/ir.cpp ../src/cuda_ir.cpp)
target_link_libraries(bspai_cuda_ir PUBLIC bspai_cuda_gmres bspai_cuda_spai)
target_compile_features(bspai_cuda_ir PUBLIC cxx_std_11)
add_executable(cuda_hybrid_ir ../src/cuda_hybrid_main.cpp)
target_link_libraries(cuda_hybrid_ir PRIVATE bspai_cuda_ir)
add_executable(cuda_gpu_spai_ir ../src/cuda_hybrid_main.cpp)
target_compile_definitions(cuda_gpu_spai_ir PRIVATE BSPAI_GPU_CONSTRUCTION=1)
target_link_libraries(cuda_gpu_spai_ir PRIVATE bspai_cuda_ir)
if(MSVC)
  target_compile_options(bspai_cuda_ir PRIVATE /W4 /fp:strict)
  target_compile_options(cuda_hybrid_ir PRIVATE /W4 /fp:strict)
  target_compile_options(cuda_gpu_spai_ir PRIVATE /W4 /fp:strict)
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test_cuda_spai.cpp")
  add_executable(cuda_spai_tests ../tests/test_cuda_spai.cpp)
  target_link_libraries(cuda_spai_tests PRIVATE bspai_cuda_ir)
  if(MSVC)
    target_compile_options(cuda_spai_tests PRIVATE /W4 /fp:strict)
  endif()
  add_test(NAME cuda_spai_tests COMMAND cuda_spai_tests)
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test_cuda_ir.cpp")
  add_executable(cuda_ir_tests ../tests/test_cuda_ir.cpp)
  target_link_libraries(cuda_ir_tests PRIVATE bspai_cuda_ir)
  if(MSVC)
    target_compile_options(cuda_ir_tests PRIVATE /W4 /fp:strict)
  endif()
  add_test(NAME cuda_ir_tests COMMAND cuda_ir_tests)
endif()
add_executable(cuda_environment_probe ../src/cuda_environment.cpp)
target_link_libraries(cuda_environment_probe PRIVATE CUDA::cudart CUDA::cusparse CUDA::cublas CUDA::cusolver)
add_executable(cuda_norm_probe ../src/cuda_norm_probe.cpp)
target_link_libraries(cuda_norm_probe PRIVATE CUDA::cudart CUDA::cublas)
if(MSVC)
  target_compile_options(cuda_norm_probe PRIVATE /W4 /fp:strict)
endif()
if(MSVC)
  target_compile_options(bspai_cuda_spmv PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/fp:strict> $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/W4,/fp:strict>)
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test_cuda_gmres.cpp")
  add_executable(cuda_gmres_tests ../tests/test_cuda_gmres.cpp)
  target_link_libraries(cuda_gmres_tests PRIVATE bspai_cuda_gmres)
  if(MSVC)
    target_compile_options(cuda_gmres_tests PRIVATE /W4 /fp:strict)
  endif()
  add_test(NAME cuda_gmres_tests COMMAND cuda_gmres_tests)
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test_cuda_spmv.cpp")
  add_executable(cuda_spmv_tests ../tests/test_cuda_spmv.cpp)
  target_link_libraries(cuda_spmv_tests PRIVATE bspai_cuda_spmv)
  if(MSVC)
    target_compile_options(cuda_spmv_tests PRIVATE /W4 /fp:strict)
  endif()
  add_test(NAME cuda_spmv_tests COMMAND cuda_spmv_tests)
endif()
